select
timestamp_micros(event_timestamp) as etkinlik_zamani,
user_pseudo_id,
(select value.int_value from unnest(event_params) where key='ga_session_id') as oturum_id,
event_name as etkinlik_adi,
geo.country as ulke,
device.category as cihaz_kategorisi,
traffic_source.source as kaynak,
traffic_source.medium as arac,
traffic_source.name as kampanya
from
`bigquery-public-data.ga4_obfuscated_sample_ecommerce.events_*`
where
_table_suffix between '20210101'and '20211231'
and
event_name in (
'session_start',
'view_item',
'add_to_cart',
'begin_checkout',
'add_shipping_info',
'add_payment_info',
'purchase'
);
